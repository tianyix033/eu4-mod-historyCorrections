# HistoryCorrections
#
# On Actions combine across files as of 1.36, so this adds to vanilla's
# 00_on_actions.txt rather than replacing it.
#
# Baseline military tradition. Army and navy tradition pivot at 50 in
# BaseModifiersTweaks, so a country at 0 carries the full penalty. Starting
# everyone at 25 puts the median at the midpoint of the penalty half and lets
# the game's own tradition gain carry them toward neutral.
#
# on_startup is country-scoped and fires once per country at game start.
# Countries formed later start from 0, which is intended -- a newly proclaimed
# state has no veterans.
#
# Countries with a war record get more on top of this, in history/countries/.

on_startup = {
	add_army_tradition = 25
	add_navy_tradition = 25
	set_country_flag = hc_claimed_starting_tradition

	# Monuments that already stood on 1556.2.14 ---------------------------
	#
	# A monument's tier is fixed by starting_tier in common/great_projects/, so
	# every start date opens with the same tiers and a 1556 world looks exactly
	# like 1444. Tier 0 is how Paradox says "unbuilt or in ruins" - Versailles,
	# the Brandenburg Gate and the White House all sit at 0 - so tier 0 is usually
	# CORRECT and must not be swept. These eight are the ones that demonstrably
	# stood by February 1556 and are wrongly still ruins.
	#
	# Deliberately NOT touched:
	#   the 85 monuments already at tier 1 or 2 - no blanket upkeep sweep
	#   buda_castle, which is really the Bibliotheca Corviniana, dispersed when
	#       the Ottomans took Buda in 1526
	#   cartagena_de_indias, a 17th-century fort system built after Drake's 1586
	#       raid, and fuerte_del_morro, a mere tower until the 1595 bastion fort
	#   the ancient ruins - Petra, the Parthenon, Stonehenge, Cheops, Tikal,
	#       Borobudur - which EU4 means you to restore yourself
	#   near-misses finished just after the date: St Basil's (1561), the Kazan
	#       kremlin (from 1556), St Peter's (1626), Kanbawzathadi Palace
	#
	# Guarded three ways: is_year so a 1444 start never sees it, owns so each one
	# fires exactly once and the owner is the country scope add_great_project_tier
	# needs, and has_great_project so re-running cannot stack tiers.
	if = {
		limit = {
			is_year = 1556
			has_dlc = "Leviathan"
		}
		if = {
			limit = { owns = 227 }
			# Belem Tower, Lisbon. Built 1514-1519.
			227 = {
				if = {
					limit = { NOT = { has_great_project = { type = belem_tower tier = 1 } } }
					add_great_project_tier = { type = belem_tower tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 514 }
			# Mehrangarh Fort, Jodhpur. Founded 1459 by Rao Jodha.
			514 = {
				if = {
					limit = { NOT = { has_great_project = { type = mehrangarh_fort tier = 1 } } }
					add_great_project_tier = { type = mehrangarh_fort tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 2310 }
			# Spiral Minaret of Samarra. Built 848-852 and still standing.
			2310 = {
				if = {
					limit = { NOT = { has_great_project = { type = spiral_minaret_samarra tier = 1 } } }
					add_great_project_tier = { type = spiral_minaret_samarra tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 795 }
			# Cerro Rico del Potosi. Silver struck 1545; by 1556 one of the
			# largest cities in the Americas.
			795 = {
				if = {
					limit = { NOT = { has_great_project = { type = potosi tier = 1 } } }
					add_great_project_tier = { type = potosi tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 856 }
			# Zacatecas. Founded 1546 on the San Bernabe lode, 34 mines
			# working by 1550.
			856 = {
				if = {
					limit = { NOT = { has_great_project = { type = zacatecas_mine_city tier = 1 } } }
					add_great_project_tier = { type = zacatecas_mine_city tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 756 }
			# Salvador da Bahia. Founded 1549 as the capital of Portuguese
			# Brazil.
			756 = {
				if = {
					limit = { NOT = { has_great_project = { type = salvador_da_bahia tier = 1 } } }
					add_great_project_tier = { type = salvador_da_bahia tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 4144 }
			# Heddal Stave Church. Early 13th century.
			4144 = {
				if = {
					limit = { NOT = { has_great_project = { type = heddal_stave_church tier = 1 } } }
					add_great_project_tier = { type = heddal_stave_church tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 1170 }
			# M'banza-Kongo, capital of Kongo; its cathedral went up 1526-1534.
			1170 = {
				if = {
					limit = { NOT = { has_great_project = { type = mbanza_kongo tier = 1 } } }
					add_great_project_tier = { type = mbanza_kongo tier = 1 }
				}
			}
		}
	}

	# Monuments that were materially built up between 1444 and 1556 ---------
	#
	# These eleven already stand at tier 1; +1 takes them to 2. Each one is here
	# because of documented construction inside the window, not because its owner
	# is large - which matters, since the owner can change. Capped at tier 2, so
	# vanilla's invariant that nothing starts at tier 3 still holds.
	#
	# Rejected from the same size-based sweep, recorded so they are not revisited:
	#   chichen_itza, machu_picchu, qhapaq_nam, tzintzuntzan - these would have
	#       upgraded only because Spain is large, when the conquest that handed
	#       them to Spain is precisely what ruined them
	#   divrigi_hospital_mosque - untouched since 1228
	#   grand_canal_1 to _4 and dujiangyan - the big Ming canal restoration was
	#       1411-15, just before the window, and four segments is a large boost
	#   citadel_of_aleppo - rebuilt after Timur, also pre-window
	#   etchimiadzin_cathedral - the Catholicosate returned in 1441, pre-window
	#   krakow_cloth_hall - it BURNED DOWN in 1555 and was rebuilt 1556-59, so on
	#       our start date it is a shell. Left at tier 1 rather than downgraded.
	if = {
		limit = {
			is_year = 1556
			has_dlc = "Leviathan"
		}
		if = {
			limit = { owns = 295 }
			# Moscow Kremlin. Its present brick walls and towers were built
			# 1485-1495 by Italian architects under Ivan III, with the Dormition
			# Cathedral 1475-79, the Faceted Chamber 1487, the Archangel
			# Cathedral 1505-08 and the Bell Tower 1508. The clearest case here.
			295 = {
				if = {
					limit = { NOT = { has_great_project = { type = kremlin tier = 2 } } }
					add_great_project_tier = { type = kremlin tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 698 }
			# Great Wall. The brick wall that survives is largely 15th-16th
			# century, extended hard under Jiajing against Altan Khan - which is
			# exactly our window and our reigning emperor.
			698 = {
				if = {
					limit = { NOT = { has_great_project = { type = the_great_wall_of_china tier = 2 } } }
					add_great_project_tier = { type = the_great_wall_of_china tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 541 }
			# Vijayanagara. At its architectural peak under Krishnadevaraya
			# 1509-29 - the Krishna temple, Vittala's eastern gopuram and ranga
			# mandapa, both 1510. Not sacked until 1565, nine years after we start.
			541 = {
				if = {
					limit = { NOT = { has_great_project = { type = hampi tier = 2 } } }
					add_great_project_tier = { type = hampi tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 4703 }
			# Rila. Destroyed in the mid-15th century and rebuilt at its end,
			# when St Ivan's relics were moved back in.
			4703 = {
				if = {
					limit = { NOT = { has_great_project = { type = rila_monasteries tier = 2 } } }
					add_great_project_tier = { type = rila_monasteries tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 223 }
			# Alhambra. Charles V began his palace inside it in 1527.
			223 = {
				if = {
					limit = { NOT = { has_great_project = { type = alhambra tier = 2 } } }
					add_great_project_tier = { type = alhambra tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 104 }
			# Milan Cathedral. Under construction throughout; the tiburio was
			# closed in 1500.
			104 = {
				if = {
					limit = { NOT = { has_great_project = { type = duomo_milano tier = 2 } } }
					add_great_project_tier = { type = duomo_milano tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 1816 }
			# Forbidden City. The three main halls burned in 1421 and were
			# rebuilt, with further heavy work under Jiajing.
			1816 = {
				if = {
					limit = { NOT = { has_great_project = { type = forbidden_city tier = 2 } } }
					add_great_project_tier = { type = forbidden_city tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 2140 }
			# Temple of Confucius, Qufu. Burned in 1499 and rebuilt by 1504.
			2140 = {
				if = {
					limit = { NOT = { has_great_project = { type = temple_of_confucius tier = 2 } } }
					add_great_project_tier = { type = temple_of_confucius tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 214 }
			# Aljaferia. Rebuilt for the Catholic Monarchs 1488-92.
			214 = {
				if = {
					limit = { NOT = { has_great_project = { type = aljaferia_palace tier = 2 } } }
					add_great_project_tier = { type = aljaferia_palace tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 361 }
			# Al-Azhar. Minarets and expansions under Qaytbay and al-Ghuri,
			# 1468-1516.
			361 = {
				if = {
					limit = { NOT = { has_great_project = { type = al_azhar_university tier = 2 } } }
					add_great_project_tier = { type = al_azhar_university tier = 1 }
				}
			}
		}
		if = {
			limit = { owns = 236 }
			# Tower of London. Henry VIII's refit of 1532-33.
			236 = {
				if = {
					limit = { NOT = { has_great_project = { type = tower_of_london tier = 2 } } }
					add_great_project_tier = { type = tower_of_london tier = 1 }
				}
			}
		}
	}
}

# A country annexed now may be released again later, at which point it should be
# able to claim again. FROM is the annexed country.
on_annexed = {
	FROM = { clr_country_flag = hc_claimed_starting_tradition }
}
