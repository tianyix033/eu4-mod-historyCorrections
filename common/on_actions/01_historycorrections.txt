# HistoryCorrections
#
# On Actions combine across files as of 1.36, so this adds to vanilla's
# 00_on_actions.txt rather than replacing it.
#
# Baseline military tradition. Army and navy tradition pivot at 50 in
# BaseModifiersTweaks, so a country at 0 carries the full penalty. Starting
# everyone at 25 puts the median at the midpoint of the penalty half and lets
# the game's own tradition gain carry them toward neutral.
#
# on_startup is country-scoped and fires once per country at game start.
# Countries formed later start from 0, which is intended -- a newly proclaimed
# state has no veterans.
#
# Countries with a war record get more on top of this, in history/countries/.

on_startup = {
	add_army_tradition = 25
	add_navy_tradition = 25

	# every start date opens with the same tiers and a 1556 world looks exactly
	# like 1444. Tier 0 is how Paradox says "unbuilt or in ruins" - Versailles,
	# the Brandenburg Gate and the White House all sit at 0 - so tier 0 is usually
	# CORRECT and must not be swept. These eight are the ones that demonstrably
	# stood by February 1556 and are wrongly still ruins.

	#   the 85 monuments already at tier 1 or 2 - no blanket upkeep sweep

	#       raid, and fuerte_del_morro, a mere tower until the 1595 bastion fort

			1816 = {

	# Age of Reformation - accumulated splendor and the objectives -------------
	#
	# Splendor income: base_values gives monthly_splendor = 1 to EVERY country,
	# num_objectives_fullfilled adds +2/month PER OBJECTIVE FULFILLED, lucky
	# nations get +1, and an ability costs 800 (SPLENDOR_ABILITY_COST).
	#
	# Splendor CARRIES across ages but abilities do NOT - on_new_age runs
	# remove_all_age_timed_modifiers and never touches splendor - so a 1444
	# campaign reaching 1556 has earned its whole income and spent nearly all of
	# it re-buying abilities after each age transition. We therefore grant income
	# for THIS AGE ONLY and treat everything earlier as already spent.
	#
	# The age is taken to have begun 1520.1.1: the file says start = 1530, but a
	# 1444 campaign usually reaches it earlier, so 1520 is the middle ground.
	# That is 433 months to 1556.2.14.
	#
	# Everyone gets the 433 months of base income. On top of that, each of the
	# seven objectives pays 2/month for as long as the country actually held it,
	# so the sections below add 2 x months per objective. Five of the seven are
	# live triggers that the engine will also evaluate at runtime, which is what
	# keeps the income flowing after the start date; what is granted here is only
	# the accumulated past that a late start cannot know about.
	#
	# obj_force_converted is the one objective granted to NOBODY. Forcing a whole
	# nation to change religion by treaty has essentially no 1444-1556 analogue,
	# and the nearest thing - the 1555 Peace of Augsburg - established the
	# opposite principle, that each prince settled his own lands.
	if = {
		limit = {
			is_year = 1556
			has_dlc = "Mandate of Heaven"
		}

		# --- obj_christian_convert

		# history/countries, so these are measured, not estimated. 'pre-age' means it

		if = {
			limit = { tag = PRU }
			add_splendor = 746
		}

		if = {
			limit = { tag = HES }
			add_splendor = 698
		}
		# 337 months (28.1 years), 2 countries
		if = {
			limit = {
				OR = {
					tag = SWI tag = TTL
				}

		}

		if = {

		}

		}

		if = {

		if = {

				OR = {

				}
			}

		}

		if = {
			limit = {
				OR = {
					tag = STE tag = WOL tag = WUR
				}
			}

		}

# A country annexed now may be released again later, at which point it should be
# able to claim again. FROM is the annexed country.

	FROM = { clr_country_flag = hc_claimed_starting_tradition }
}
